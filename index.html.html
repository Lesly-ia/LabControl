Configuración Directa de Supabase

Conecta tu proyecto con las columnas en español

SQL para crear la tabla en Supabase: 
create table public.reservas (
  id uuid default gen_random_uuid() primary key,
  estudiante text not null,
  equipo text not null,
  fecha_reserva date not null,
  hora_inicio text not null,
  created_at timestamp with time zone default now() not null,
  unique (equipo, fecha_reserva, hora_inicio)
);

alter table public.reservas enable row level security;
create policy "Lectura pública" on public.reservas for select using (true);
create policy "Inserción pública" on public.reservas for insert with check (true);
create policy "Eliminación pública" on public.reservas for delete using (true);
